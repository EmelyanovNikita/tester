# 1. SDP для первого канала
sdp_offer_1 = "v=0\r\no=SIPP-PERFORMANCE-TEST 123456 654321 IN IP4 10.30.100.108\r\ns=SIPP-RTP-session\r\nc=IN IP4 10.30.100.108\r\nt=0 0\r\nm=audio 6000 RTP/AVP 8 0\r\na=rtpmap:8 PCMA/8000/1\r\na=rtpmap:0 PCMU/8000/1\r\na=ptime:20\r\na=sendrecv\r\n"

# 2. Создаём первый канал
resp = ChannelAdd(SESSION_ID, 0, 0, true, sdp_offer_1, 0, "")

channel_id_1 = resp.channelId
context_id = resp.contextId

# 3. DTMF детектор на первом канале
dtmf_settings = {"type": 1, "pass_through_mode": false}
resp = ChannelDTMFDetectStart(SESSION_ID, channel_id_1, dtmf_settings)

# 4. Второй канал (с медиа)
resp = ChannelAdd(SESSION_ID, context_id, 0, true, "", 0, "")
channel_id_2 = resp.channelId

# 5. Тон на первом канале (числа как строки)
tone_info = {"frequency1": "425", "frequency2": "425", "onMs": "1000", "offMs": "4000", "duration": "0"}
resp = ChannelTonePlay(SESSION_ID, channel_id_1, tone_info)

# 6. Запрос ресурса для второго канала (SDP как строка)
port_info = {"action": "ACT_MODIFY", "sdp_answer": "v=0\r\no=- 1 11 IN IP4 10.30.100.108\r\ns=-\r\nc=IN IP4 10.30.100.108\r\nt=0 0\r\nm=audio 6920 RTP/AVP 8 96\r\na=rtpmap:8 PCMA/8000/1\r\na=rtpmap:96 telephone-event/8000\r\na=fmtp:96 0-15\r\n", "sdp_offer": ""}
resp = ChannelResource(SESSION_ID, channel_id_2, port_info, "")

# 8. DTMF детектор на втором канале
resp = ChannelDTMFDetectStart(SESSION_ID, channel_id_2, dtmf_settings)

# 9. Обновляем второй канал
resp = ChannelAdd(SESSION_ID, context_id, channel_id_2, false, "", 0, "")

# 10. Обновляем первый канал
resp = ChannelAdd(SESSION_ID, context_id, channel_id_1, false, "", 0, "")
