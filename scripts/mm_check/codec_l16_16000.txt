# Проверка на реальном MM: L16 16 кГц (linear PCM)
# Тестер предлагает только этот кодек, MM должен его принять, прислать RTP и не терять пакеты.
# Запуск: ./tester -p config/config.json -s scripts/mm_check/codec_l16_16000.txt   (или всей папкой: -s scripts/mm_check)
# Результат: сценарий завершается успешно (в логе «ОК l16_16000») или ошибкой «ПРОВАЛ ...»; код возврата 1.
# Если MM не принял кодек, ошибка будет уже на ChannelAdd (reply_code не RC_SUCCESS).
rtp = rtp_open(codecs = "L16/16000")
chan = ChannelAdd(SESSION_ID, 0, 0, true, rtp.sdp, 0, "")

rtp_start(rtp, chan.sdp_answer, tone = 440, record = "mm_check_l16_16000_{id}.wav")
sleep(3)
st = rtp_stop(rtp)
ChannelRemove(SESSION_ID, chan.channel_id)

log("l16_16000: кодек", st.codec, "отправлено", st.sent_packets, "принято", st.received_packets, "потеряно", st.lost_packets, "джиттер, мс", st.jitter_ms)

if st.codec != "L16/16000/1" {
    fail("ПРОВАЛ l16_16000: MM выбрал кодек", st.codec, "вместо L16/16000/1")
}

if st.received_packets < 20 {
    fail("ПРОВАЛ l16_16000: от MM пришло слишком мало RTP-пакетов:", st.received_packets)
}

if st.lost_packets > 5 {
    fail("ПРОВАЛ l16_16000: потеряно пакетов:", st.lost_packets)
}

log("ОК l16_16000")
