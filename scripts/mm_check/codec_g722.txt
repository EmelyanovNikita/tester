# Проверка на реальном MM: G.722 (16 кГц)
# Тестер предлагает только этот кодек, MM должен его принять, прислать RTP и не терять пакеты.
# Запуск: ./tester -p config/config.json -s scripts/mm_check/codec_g722.txt   (или всей папкой: -s scripts/mm_check)
# Результат: сценарий завершается успешно (в логе «ОК g722») или ошибкой «ПРОВАЛ ...»; код возврата 1.
# Если MM не принял кодек, ошибка будет уже на ChannelAdd (reply_code не RC_SUCCESS).
rtp = rtp_open(codecs = "G722")
chan = ChannelAdd(SESSION_ID, 0, 0, true, rtp.sdp, 0, "")

rtp_start(rtp, chan.sdp_answer, tone = 440, record = "mm_check_g722_{id}.wav")
sleep(3)
st = rtp_stop(rtp)
ChannelRemove(SESSION_ID, chan.channel_id)

log("g722: кодек", st.codec, "отправлено", st.sent_packets, "принято", st.received_packets, "потеряно", st.lost_packets, "джиттер, мс", st.jitter_ms)

if st.codec != "G722" {
    fail("ПРОВАЛ g722: MM выбрал кодек", st.codec, "вместо G722")
}

if st.received_packets < 20 {
    fail("ПРОВАЛ g722: от MM пришло слишком мало RTP-пакетов:", st.received_packets)
}

if st.lost_packets > 5 {
    fail("ПРОВАЛ g722: потеряно пакетов:", st.lost_packets)
}

log("ОК g722")
