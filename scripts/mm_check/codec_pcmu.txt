# Проверка на реальном MM: PCMU (G.711 µ-law)
# Тестер предлагает только этот кодек, MM должен его принять, прислать RTP и не терять пакеты.
# Запуск: ./tester -p config/config.json -s scripts/mm_check/codec_pcmu.txt   (или всей папкой: -s scripts/mm_check)
# Результат: сценарий завершается успешно (в логе «ОК pcmu») или ошибкой «ПРОВАЛ ...»; код возврата 1.
# Если MM не принял кодек, ошибка будет уже на ChannelAdd (reply_code не RC_SUCCESS).
rtp = rtp_open(codecs = "PCMU")
chan = ChannelAdd(SESSION_ID, 0, 0, true, rtp.sdp, 0, "")

rtp_start(rtp, chan.sdpAnswer, tone = 440, record = "mm_check_pcmu_{id}.wav")
sleep(3)
st = rtp_stop(rtp)
ChannelRemove(SESSION_ID, chan.channelId)

log("pcmu: кодек", st.codec, "отправлено", st.sentPackets, "принято", st.receivedPackets, "потеряно", st.lostPackets, "джиттер, мс", st.jitterMs)

if st.codec != "PCMU" {
    fail("ПРОВАЛ pcmu: MM выбрал кодек", st.codec, "вместо PCMU")
}

if st.receivedPackets < 20 {
    fail("ПРОВАЛ pcmu: от MM пришло слишком мало RTP-пакетов:", st.receivedPackets)
}

if st.lostPackets > 5 {
    fail("ПРОВАЛ pcmu: потеряно пакетов:", st.lostPackets)
}

log("ОК pcmu")
