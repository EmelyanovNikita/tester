# Проверка на реальном MM: PCMA с ptime 10 мс
# Тестер предлагает только этот кодек, MM должен его принять, прислать RTP и не терять пакеты.
# Запуск: ./tester -p config/config.json -s scripts/mm_check/codec_pcma_ptime10.txt   (или всей папкой: -s scripts/mm_check)
# Результат: сценарий завершается успешно (в логе «ОК pcma_ptime10») или ошибкой «ПРОВАЛ ...»; код возврата 1.
# Если MM не принял кодек, ошибка будет уже на ChannelAdd (reply_code не RC_SUCCESS).
rtp = rtp_open(codecs = "PCMA", ptime = 10)
chan = ChannelAdd(SESSION_ID, 0, 0, true, rtp.sdp, 0, "")

rtp_start(rtp, chan.sdp_answer, tone = 440, record = "mm_check_pcma_ptime10_{id}.wav")
sleep(3)
st = rtp_stop(rtp)
ChannelRemove(SESSION_ID, chan.channel_id)

log("pcma_ptime10: кодек", st.codec, "отправлено", st.sent_packets, "принято", st.received_packets, "потеряно", st.lost_packets, "джиттер, мс", st.jitter_ms)

if st.codec != "PCMA" {
    fail("ПРОВАЛ pcma_ptime10: MM выбрал кодек", st.codec, "вместо PCMA")
}

if st.received_packets < 20 {
    fail("ПРОВАЛ pcma_ptime10: от MM пришло слишком мало RTP-пакетов:", st.received_packets)
}

if st.lost_packets > 5 {
    fail("ПРОВАЛ pcma_ptime10: потеряно пакетов:", st.lost_packets)
}

log("ОК pcma_ptime10")
